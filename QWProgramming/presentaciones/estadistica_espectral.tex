\newcommand{\CoinHeading}[1]{%
  \begin{minipage}[c][0.43cm][c]{0.30\textwidth}
    \strut
  \end{minipage}%
}
\newcommand{\PlotCell}[3]{%
  \begin{minipage}[c][3.85cm][c]{0.30\textwidth}
    \centering
    \IfFileExists{Imgs/#1_#2_#3.png}{%
      \includegraphics[width=\linewidth,height=3.75cm,keepaspectratio]{Imgs/#1_#2_#3.png}%
    }{%
      \IfFileExists{Imgs/#1_#2_#3.pdf}{%
        \includegraphics[width=\linewidth,height=3.75cm,keepaspectratio,clip]{Imgs/#1_#2_#3.pdf}%
      }{%
        \colorbox{gray!10}{\parbox[c][3.55cm][c]{0.92\linewidth}{%
          \centering\small\color{gray!65!black}Sin figura guardada\\en el notebook}}%
      }%
    }%
  \end{minipage}%
}
\newcommand{\StadiumRow}[6]{%
  \noindent
  \begin{minipage}[c][3.85cm][c]{1.28cm}
    \strut
  \end{minipage}\hfill
  \PlotCell{#1}{#2}{#4}\hfill
  \PlotCell{#1}{#2}{#5}\hfill
  \PlotCell{#1}{#2}{#6}\par
}
\newcommand{\GridPage}[5]{%
  \begin{center}\large\bfseries\phantom{#1}\end{center}
  \noindent\hspace*{1.28cm}\hfill
  \CoinHeading{#3}\hfill\CoinHeading{#4}\hfill\CoinHeading{#5}\par
  \vspace{0.05cm}
  \StadiumRow{#2}{Rect}{Rect.}{#3}{#4}{#5}
  \vspace{0.08cm}
  \StadiumRow{#2}{Sinai}{Sinaí}{#3}{#4}{#5}
  % Los rótulos se trazan al final para que queden por encima de las figuras.
  \begin{tikzpicture}[remember picture,overlay]
    \node at ([yshift=-0.78cm]current page.north) {\large\bfseries #1};
    \node at ([xshift=4.87cm,yshift=-1.59cm]current page.north west)
      {\small\bfseries\MakeUppercase{#3}};
    \node at ([xshift=10.74cm,yshift=-1.59cm]current page.north west)
      {\small\bfseries\MakeUppercase{#4}};
    \node at ([xshift=16.61cm,yshift=-1.59cm]current page.north west)
      {\small\bfseries\MakeUppercase{#5}};
    \node at ([xshift=1.18cm,yshift=-3.85cm]current page.north west)
      {\small\bfseries Rect.};
    \node at ([xshift=1.18cm,yshift=-7.75cm]current page.north west)
      {\small\bfseries Sinaí};
  \end{tikzpicture}%
  \newpage
}
