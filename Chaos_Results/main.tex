\section{Discrete-time quantum walks in billiards} % {{{*
\label{sec:dtqw-billiards}

A discrete-time quantum walk (DTQW) describes a walker moving on a lattice 
in discrete time steps, with its motion controlled by an internal degree of 
freedom called the coin. The coin basis states encode the possible directions 
of motion. Here, we consider a lattice confined to a billiard geometry, with 
four coin states corresponding to right, left, up, and down.
Let $\Omega\subset\mathbb Z^2$ denote the lattice sites inside the billiard 
and $N=|\Omega|$ the number of sites. The position and coin Hilbert spaces 
are, respectively,
$\mathcal H_p=\operatorname{span}\{\ket{v}:v\in\Omega\}\simeq\mathbb C^N$
and
$\mathcal H_c=\operatorname{span}\{\ket{R},\ket{L},\ket{U},\ket{D}\}\simeq\mathbb C^4$.
The walker's state belongs to $\mathcal H=\mathcal H_c\otimes\mathcal H_p$, which has dimension $4N$ and basis $\ket{c,v}=\ket{c}\otimes\ket{v}$. At step $t$, it can be written as
\begin{equation}
    \ket{\psi_t}=\sum_{v\in\Omega}\sum_c
        \psi_c(v,t)\ket{c,v},
    \qquad \sum_{v,c}|\psi_c(v,t)|^2=1,
    \label{eq:walk-state-expansion}
\end{equation}
where $c\in\{R,L,U,D\}$ and $\psi_c(v,t)$ is the amplitude for direction $c$ at site $v$.

Each DTQW step first applies the coin to create a superposition of
directions at each site, then shifts the resulting amplitudes in those
directions. The one-step evolution operator is therefore
\begin{equation}
    U=S\qty(C\otimes\mathbb{I}_p),
\end{equation}
where $C$ acts only on the coin space and $\mathbb{I}_p$ is the identity
on the position space. The shift $S$ moves each amplitude in its coin
direction. The corresponding displacements are
$\boldsymbol\delta_R=(1,0)$, $\boldsymbol\delta_L=(-1,0)$,
$\boldsymbol\delta_U=(0,1)$, and $\boldsymbol\delta_D=(0,-1)$.
Let $\overline c$ denote the direction opposite to $c$.
We impose reflecting boundary conditions: when the walker encounters
a boundary, its coin direction is reversed and its position remains
unchanged. Hence,
\begin{equation}
    S\ket{c,v}=
    \begin{cases}
        \ket{c,v+\boldsymbol\delta_c},
            &v+\boldsymbol\delta_c\in\Omega,\\
        \ket{\overline c,v},
            &v+\boldsymbol\delta_c\notin\Omega.
    \end{cases}
    \label{eq:walk-shift-action}
\end{equation}
Equivalently,
\begin{equation}
    S=\sum_c\left[
        \sum_{\substack{v\in\Omega\\v+\boldsymbol\delta_c\in\Omega}}
        \ketbra{c}{c}\otimes\ketbra{v+\boldsymbol\delta_c}{v}
        +\sum_{\substack{v\in\Omega\\v+\boldsymbol\delta_c\notin\Omega}}
        \ketbra{\overline c}{c}\otimes\ketbra{v}{v}
    \right].
    \label{eq:walk-shift-operator}
\end{equation}
The shift is unitary because it permutes the position--coin basis states.
Since the same coin is used at every site and every time, the billiard geometry enters
the evolution operator $U$ entirely through $S$, which determines both
the available displacements and the reflections at the boundary.

\section{Distribución temporal promedio} % {{{*
Sea $\ket{\alpha_t}=U^t\ket{\alpha_0}$ el estado del caminante después de
$t$ pasos. La probabilidad de encontrarlo en el nodo $v$, sin distinguir su
estado interno de moneda, es~\cite{aharonov2001quantum}
\begin{equation}
    P_t(v\mid\alpha_0)=\sum_{k=1}^{4}
    \left|\braket{c_k,v}{\alpha_t}\right|^{2},
    \label{eq:SiteProb}
\end{equation}
con $\{\ket{c_k}\}_{k=1}^{4}$ una base del espacio de moneda.

La distribución temporal promedio durante los primeros $T$ pasos se define
como~\cite{aharonov2001quantum}
\begin{equation}
    \overline{P}_{T}(v\mid\alpha_0)
    =\frac{1}{T}\sum_{t=0}^{T-1}P_t(v\mid\alpha_0)
    =\frac{1}{T}\sum_{t=0}^{T-1}\sum_{k=1}^{4}
    \left|\matrixel{c_k,v}{U^t}{\alpha_0}\right|^2.
    \label{eq:TimeAverageDist}
\end{equation}

\subsection{Comparación del IPR espacial} % {{{*
Con la probabilidad de posición $P_t(v\mid\alpha_0)$ definida en
\eqref{eq:SiteProb}, el IPR espacial en el paso $t$ es
\begin{equation}
    \mathrm{IPR}^{\mathrm{esp}}(t\mid\alpha_0)
    =\frac{\displaystyle\sum_{v\in\Omega}P_t(v\mid\alpha_0)^2}
           {\displaystyle\left(\sum_{v\in\Omega}P_t(v\mid\alpha_0)\right)^2}
    =\sum_{v\in\Omega}P_t(v\mid\alpha_0)^2
    \quad\text{si }\braket{\alpha_t}{\alpha_t}=1.
    \label{eq:ipr-spatial}
\end{equation}
Para una distribución uniforme en $N$ nodos, $\mathrm{IPR}^{\mathrm{esp}}=1/N$.

\section{Densidad espacial del IPR} % {{{*
Para un eigenestado $\ket{\phi_n}$ del operador de evolución, su
probabilidad en el nodo $v$ es
\begin{equation}
    p_n(v)=\sum_{c=1}^{4}\left|\braket{c,v}{\phi_n}\right|^2.
\end{equation}
La contribución de ese nodo al IPR espacial del eigenestado se define como
\begin{equation}
    I_n(v)=\frac{p_n(v)^2}
    {\left(\sum_{u\in\Omega}p_n(u)\right)^2},
    \qquad
    \mathrm{IPR}^{\mathrm{esp}}_n=\sum_{v\in\Omega}I_n(v).
    \label{eq:ipr-density}
\end{equation}
Para un eigenestado normalizado, $I_n(v)=p_n(v)^2$.

\section{Concentración de la medida de la pureza de Choi} % {{{*
\janote{Esto surgió de una conversación con ChatGPT, aún no está digerido por mí}

Consideremos una DTQW con
\begin{equation}
    \mathcal H=\mathcal H_c\otimes\mathcal H_p,
    \qquad
    \dim\mathcal H_c=4,
    \qquad
    \dim\mathcal H_p=N,
\end{equation}
donde $N$ es el número de sitios permitidos por la geometría del billar.
La evolución $U=SC$ se mantiene fija y la única variable será el estado
inicial de posición
\begin{equation}
    \ket{\psi}_p=\sum_{a=1}^{N}\psi_a\ket{a}_p,
    \qquad
    \braket{\psi}{\psi}=1.
\end{equation}
Al tiempo $t$, el canal reducido de la moneda es
\begin{equation}
    \mathcal E_{\psi,t}(\rho_c)
    =
    \operatorname{Tr}_p\left[
        U^t\left(\rho_c\otimes\ket{\psi}\bra{\psi}\right)U^{-t}
    \right].
\end{equation}

Usaremos la matriz de Choi normalizada
\begin{equation}
    J_{\psi,t}
    =
    (\mathbb I_r\otimes\mathcal E_{\psi,t})(\Phi_4),
    \qquad
    \Phi_4=\ket{\Phi_4}\bra{\Phi_4},
    \qquad
    \ket{\Phi_4}
    =
    \frac{1}{2}\sum_{\mu=1}^{4}
    \ket{\mu}_r\ket{\mu}_c,
\end{equation}
de modo que $\operatorname{Tr}J_{\psi,t}=1$. La cantidad de interés es
\begin{equation}
    \mathcal P_t(\psi)
    =
    \operatorname{Tr}\left(J_{\psi,t}^2\right).
    \label{eq:ChoiPurityConcentration}
\end{equation}

\subsection{Constante de Lipschitz}

Es útil introducir el canal
\begin{equation}
    \Lambda_t(\rho_p)
    =
    \operatorname{Tr}_p\left[
        (\mathbb I_r\otimes U^t)
        (\Phi_4\otimes\rho_p)
        (\mathbb I_r\otimes U^{-t})
    \right],
\end{equation}
que es CPTP y satisface
$J_{\psi,t}=\Lambda_t(\ket{\psi}\bra{\psi})$.
Definamos
\begin{equation}
    B_{\psi,t}
    =
    \Lambda_t^\dagger(J_{\psi,t}).
\end{equation}
Como $\Lambda_t^\dagger$ es unital y positivo,
$0\leq B_{\psi,t}\leq\mathbb I$, y además
\begin{equation}
    \bra{\psi}B_{\psi,t}\ket{\psi}
    =
    \operatorname{Tr}(J_{\psi,t}^2)
    =
    \mathcal P_t(\psi).
\end{equation}

Para una variación tangente $\ket{\delta\psi}$ sobre la esfera de estados,
\begin{equation}
    \delta\mathcal P_t
    =
    4\,\operatorname{Re}
    \bra{\delta\psi}B_{\psi,t}\ket{\psi},
\end{equation}
por lo que el gradiente tangencial es
\begin{equation}
    \nabla_S\mathcal P_t
    =
    4\left(B_{\psi,t}-\mathcal P_t(\psi)\mathbb I\right)\ket{\psi}.
\end{equation}
Por tanto, la constante de Lipschitz que aparece en el lema de Lévy es
exactamente
\begin{equation}
    \boxed{
    \eta_t
    =
    4\sup_{\psi}
    \sqrt{
        \bra{\psi}B_{\psi,t}^2\ket{\psi}
        -\mathcal P_t(\psi)^2
    }}.
    \label{eq:ExactLipschitzChoi}
\end{equation}
Su valor depende, en general, de $U$ y de $t$. Sin embargo,
$B_{\psi,t}^2\leq B_{\psi,t}$ implica
\begin{equation}
    \left\|\nabla_S\mathcal P_t\right\|^2
    \leq
    16\mathcal P_t(1-\mathcal P_t)
    \leq4,
\end{equation}
y obtenemos la cota universal
\begin{equation}
    \boxed{\eta_t\leq2.}
    \label{eq:LipschitzBoundChoi}
\end{equation}
La igualdad no tiene por qué alcanzarse; por ejemplo, si el canal de la
moneda es independiente de $\ket{\psi}$, entonces $\eta_t=0$.

\subsection{Promedio Haar exacto}

Sea $W_t=U^t$ y definamos los bloques de $W_t$ sobre la base de posición,
\begin{equation}
    K_{xa}^{(t)}
    =
    {}_p\!\bra{x}W_t\ket{a}_p,
\end{equation}
que son operadores $4\times4$ sobre la moneda. Para
$\ket{\psi}=\sum_a\psi_a\ket{a}$, los operadores de Kraus del canal
reducido son
\begin{equation}
    K_x(\psi)=\sum_a\psi_a K_{xa}^{(t)}.
\end{equation}
La pureza de Choi puede escribirse como
\begin{equation}
    \mathcal P_t(\psi)
    =
    \frac{1}{16}
    \sum_{x,y}
    \left|
        \operatorname{Tr}_c
        \left[K_x^\dagger(\psi)K_y(\psi)\right]
    \right|^2.
    \label{eq:ChoiPurityKraus}
\end{equation}
Introduciendo las matrices $N\times N$
\begin{equation}
    (M_{xy}^{(t)})_{ab}
    =
    \operatorname{Tr}_c
    \left[
        (K_{xa}^{(t)})^\dagger K_{yb}^{(t)}
    \right],
\end{equation}
se tiene
$\operatorname{Tr}[K_x^\dagger K_y]
=\bra{\psi}M_{xy}^{(t)}\ket{\psi}$.
Para un estado Haar aleatorio en $\mathbb C^N$,
\begin{equation}
    \mathbb E_{\psi}
    \left[
        \left|\bra{\psi}M\ket{\psi}\right|^2
    \right]
    =
    \frac{
        \operatorname{Tr}(MM^\dagger)
        +|\operatorname{Tr}M|^2
    }{N(N+1)}.
\end{equation}
Además, la unitariedad de $W_t$ implica
\begin{equation}
    \operatorname{Tr}M_{xy}^{(t)}
    =
    4\delta_{xy}.
\end{equation}
Por consiguiente, el promedio Haar de la pureza es exactamente
\begin{equation}
    \boxed{
    \overline{\mathcal P}_t
    =
    \mathbb E_{\psi}\mathcal P_t(\psi)
    =
    \frac{1}{N+1}
    +
    \frac{1}{16N(N+1)}
    \sum_{x,y}
    \operatorname{Tr}
    \left[
        M_{xy}^{(t)}(M_{xy}^{(t)})^\dagger
    \right].
    }
    \label{eq:HaarAverageChoiPurity}
\end{equation}
Esta expresión es exacta para cualquier geometría, moneda y tiempo; toda
la dependencia dinámica está contenida en los bloques de $U^t$.

\subsection{Concentración}

Los estados puros de posición corresponden a la esfera real
$S^{2N-1}$. Usando la versión del lema de Lévy
\begin{equation}
    \Pr\left(
        |f-\mathbb Ef|\geq\epsilon
    \right)
    \leq
    2\exp\left[
        -\frac{2C(d+1)\epsilon^2}{\eta^2}
    \right],
    \qquad
    C=\frac{1}{18\pi^3}
    \simeq1.79175\times10^{-3},
\end{equation}
con $d=2N-1$, obtenemos
\begin{equation}
    \boxed{
    \Pr\left(
        |\mathcal P_t(\psi)-\overline{\mathcal P}_t|
        \geq\epsilon
    \right)
    \leq
    2\exp\left[
        -\frac{4CN\epsilon^2}{\eta_t^2}
    \right].
    }
\end{equation}
Finalmente, usando la cota universal $\eta_t\leq2$,
\begin{equation}
    \boxed{
    \Pr\left(
        |\mathcal P_t(\psi)-\overline{\mathcal P}_t|
        \geq\epsilon
    \right)
    \leq
    2\exp\left[
        -\frac{N\epsilon^2}{18\pi^3}
    \right].
    }
    \label{eq:ChoiPurityLevy}
\end{equation}
Así, para $N\gg1$, la pureza de Choi de casi todos los estados iniciales
Haar de posición se concentra alrededor del valor
\eqref{eq:HaarAverageChoiPurity}, con fluctuaciones típicas que decrecen
como $N^{-1/2}$. La constante numérica de Lévy es conservadora, por lo que
la concentración observada numéricamente puede ser considerablemente más
fuerte que esta cota rigurosa.

\section{Exact spectrum of the Grover walk in a rectangular billiard} % {{{*
\label{app:grover-rectangle-spectrum}

Consider $N=L_xL_y$ sites, with $0\leq x<L_x$ and $0\leq y<L_y$.
The coin basis is $\{\ket{R},\ket{L},\ket{U},\ket{D}\}$, and
\begin{equation}
 U=S(G_4\otimes\mathbb I_p),\qquad
 G_4=2\ketbra{s}{s}-\mathbb I_c,\qquad
 \ket{s}=\tfrac12(\ket{R}+\ket{L}+\ket{U}+\ket{D}).
 \label{eq:appendix-grover-def}
\end{equation}
The shift moves one site in the coin direction; at a wall it reverses that
direction and leaves the position fixed. We seek eigenstates
$U\ket{\Psi}=z\ket{\Psi}$. Write
\begin{equation}
 \ket{\Psi}=\sum_{x,y}\ket{\Psi_{x,y}}\otimes\ket{x,y},\qquad
 \ket{\Psi_{x,y}}=\sum_{c=R,L,U,D}\psi_c(x,y)\ket{c},\qquad
 \psi_c(x,y)=\braket{c,x,y}{\Psi},
 \label{eq:appendix-grover-amplitudes}
\end{equation}
where $\ket{c,x,y}=\ket{c}\otimes\ket{x,y}$.

\subsection{From the eigenvalue equation to the dispersion relation}

The coin acts on each local ket as
\[
 G_4\ket{\Psi_{x,y}}
 =\frac12\left(\sum_c\psi_c(x,y)\right)
             (\ket{R}+\ket{L}+\ket{U}+\ket{D})
  -\ket{\Psi_{x,y}}.
\]
To write its four coefficients compactly, let
$h(x,y)=\psi_R(x,y)+\psi_L(x,y)-\psi_U(x,y)-\psi_D(x,y)$.
Regrouping the terms above gives
\begin{align}
 G_4\ket{\Psi_{x,y}}={}&
  (\psi_L-h/2)\ket{R}+(\psi_R-h/2)\ket{L}\notag\\
 &+(\psi_D+h/2)\ket{U}+(\psi_U+h/2)\ket{D},
 \label{eq:appendix-grover-coin-action}
\end{align}
with all coefficients evaluated at $(x,y)$.

The shift now identifies which local coin coefficient reaches a given
position--coin state. On an interior horizontal bond,
$S\ket{R,x,y}=\ket{R,x+1,y}$ and
$S\ket{L,x+1,y}=\ket{L,x,y}$. Projecting
$U\ket{\Psi}=z\ket{\Psi}$ onto these two destination states therefore gives
\begin{align}
 z\psi_R(x+1,y)
 &=\matrixel{R,x+1,y}{U}{\Psi}
  =\matrixel{R}{G_4}{\Psi_{x,y}}
  =\psi_L(x,y)-\tfrac12h(x,y),\notag\\
 z\psi_L(x,y)
 &=\matrixel{L,x,y}{U}{\Psi}
  =\matrixel{L}{G_4}{\Psi_{x+1,y}}
  =\psi_R(x+1,y)-\tfrac12h(x+1,y).
 \label{eq:appendix-grover-bond}
\end{align}
The same projections on a vertical bond give
\begin{equation}
 z\psi_U(x,y+1)=\psi_D(x,y)+\tfrac12h(x,y),\qquad
 z\psi_D(x,y)=\psi_U(x,y+1)+\tfrac12h(x,y+1).
 \label{eq:appendix-grover-vertical-bond}
\end{equation}
At the left wall, $S\ket{L,0,y}=\ket{R,0,y}$ instead, so
\[
 z\psi_R(0,y)=\matrixel{L}{G_4}{\Psi_{0,y}}
             =\psi_R(0,y)-\tfrac12h(0,y).
\]
The four wall conditions are consequently
\begin{align}
 (1-z)\psi_R(0,y)&=\tfrac12h(0,y),&
 (1-z)\psi_L(L_x-1,y)&=\tfrac12h(L_x-1,y),\notag\\
 (1-z)\psi_U(x,0)&=-\tfrac12h(x,0),&
 (1-z)\psi_D(x,L_y-1)&=-\tfrac12h(x,L_y-1).
 \label{eq:appendix-grover-walls}
\end{align}

For $z\neq\pm1$, eliminate one amplitude from each bond pair. For example,
multiplying the first equation in \eqref{eq:appendix-grover-bond} by $z$
and using the second gives
$2(1-z^2)\psi_R(x+1,y)=h(x+1,y)+zh(x,y)$.
Doing this for the four directions yields
\begin{align}
 \psi_R(x,y)&=\frac{h(x,y)+zh(x-1,y)}{2(1-z^2)},&
 \psi_L(x,y)&=\frac{h(x,y)+zh(x+1,y)}{2(1-z^2)},\notag\\
 \psi_U(x,y)&=-\frac{h(x,y)+zh(x,y-1)}{2(1-z^2)},&
 \psi_D(x,y)&=-\frac{h(x,y)+zh(x,y+1)}{2(1-z^2)}.
 \label{eq:appendix-grover-components}
\end{align}
The wall conditions are included by extending $h$ with reflected values:
\begin{equation}
 h(-1,y)=h(0,y),\quad h(L_x,y)=h(L_x-1,y),\quad
 h(x,-1)=h(x,0),\quad h(x,L_y)=h(x,L_y-1).
 \label{eq:appendix-grover-endpoints}
\end{equation}
For instance, the first expression in
\eqref{eq:appendix-grover-components} at $x=0$ becomes
$\psi_R(0,y)=h(0,y)/[2(1-z)]$, exactly as required by
\eqref{eq:appendix-grover-walls}.

Substituting \eqref{eq:appendix-grover-components} into the definition of
$h$ gives, before rearranging,
\[
 2(1-z^2)h(x,y)
 =4h(x,y)+z\big[h(x-1,y)+h(x+1,y)+h(x,y-1)+h(x,y+1)\big].
\]
Thus the eigenvalue problem reduces to the scalar equation
\begin{equation}
 h(x-1,y)+h(x+1,y)+h(x,y-1)+h(x,y+1)
 =-2(z+z^{-1})h(x,y).
 \label{eq:appendix-grover-reduced}
\end{equation}
Here $h$ cannot vanish identically, since
\eqref{eq:appendix-grover-components} would then give $\ket{\Psi}=0$.

To solve this equation, separate the spatial dependence. In one direction,
$X(x)=\cos[k(x+\tfrac12)]$ satisfies $X(-1)=X(0)$.
The other endpoint requires
$X(L)-X(L-1)=-2\sin(kL)\sin(k/2)=0$, giving $k=\pi m/L$.
The values $m=0,\ldots,L-1$ give the $L$ independent modes.
Accordingly, the normalized product modes are
\begin{equation}
 \phi_{mn}(x,y)=
 \sqrt{\frac{(2-\delta_{m0})(2-\delta_{n0})}{N}}\,
 \cos\!\left[\frac{\pi m}{L_x}\left(x+\tfrac12\right)\right]
 \cos\!\left[\frac{\pi n}{L_y}\left(y+\tfrac12\right)\right],
 \label{eq:appendix-grover-spatial-modes}
\end{equation}
with $0\leq m<L_x$ and $0\leq n<L_y$.
The cosine addition formula gives horizontal and vertical neighbor sums
$2\cos(\pi m/L_x)\phi_{mn}$ and $2\cos(\pi n/L_y)\phi_{mn}$.
Taking $h$ proportional to $\phi_{mn}$ in
\eqref{eq:appendix-grover-reduced} therefore yields
\begin{equation}
 z+z^{-1}=-2a_{mn},\qquad z^2+2a_{mn}z+1=0,\qquad
 a_{mn}=\tfrac12\left[\cos\!\left(\frac{\pi m}{L_x}\right)
                    +\cos\!\left(\frac{\pi n}{L_y}\right)\right].
 \label{eq:appendix-grover-secular}
\end{equation}
This establishes the dispersion relation and its two explicit eigenvalues:
\begin{equation}
 z_{mn}^{\pm}=-a_{mn}\pm i\sqrt{1-a_{mn}^2}
             =e^{\pm i\omega_{mn}},\qquad
 \omega_{mn}=\arccos(-a_{mn}),\qquad (m,n)\neq(0,0).
 \label{eq:appendix-grover-dispersion}
\end{equation}
Only the constant mode has $a_{00}=1$; it must be treated separately because
the reconstruction \eqref{eq:appendix-grover-components} excludes $z=-1$.

For every other mode, choose
$h=(1-z^2)\phi_{mn}/\sqrt{2(1-a_{mn}^2)}$ in
\eqref{eq:appendix-grover-components}. The resulting normalized eigenket,
with $z=z_{mn}^{\pm}$ and $\phi=\phi_{mn}$, is
\begin{align}
 \ket{\Psi_{mn}^{\pm}}
 =\frac{1}{2\sqrt{2(1-a_{mn}^2)}}\sum_{x,y}\big\{&
  [\phi(x,y)+z\phi(x-1,y)]\ket{R}
 +[\phi(x,y)+z\phi(x+1,y)]\ket{L}\notag\\
 &-[\phi(x,y)+z\phi(x,y-1)]\ket{U}
  -[\phi(x,y)+z\phi(x,y+1)]\ket{D}\big\}\otimes\ket{x,y}.
 \label{eq:appendix-grover-eigenvectors}
\end{align}
The reflected values of $\phi$ make this expression valid at walls and
corners. The bond equations verify
$U\ket{\Psi_{mn}^{\pm}}=z_{mn}^{\pm}\ket{\Psi_{mn}^{\pm}}$, and the squared
norm of the numerator is $8(1-a_{mn}^2)$, using
$\sum_{x,y}|\phi_{mn}|^2=1$ and $\operatorname{Re}z=-a_{mn}$.
This eigenket remains valid when the two cosines coincide.

\subsection{Eigenvalues \texorpdfstring{$\pm1$}{+1 and -1} and completeness}

At $z=\pm1$, the bond equations imply $h(x+1,y)=-zh(x,y)$ and
$h(x,y+1)=-zh(x,y)$. For $z=+1$, \eqref{eq:appendix-grover-walls} forces
$h=0$ at the boundary, and the bond relations propagate this condition
throughout the rectangle. For $z=-1$, $h$ is constant. Its nonzero part is
represented by
\begin{equation}
 \ket{\Psi_0^-}=\frac{1}{2\sqrt N}\sum_{x,y}
  (\ket{R}+\ket{L}-\ket{U}-\ket{D})\otimes\ket{x,y},
 \qquad U\ket{\Psi_0^-}=-\ket{\Psi_0^-}.
 \label{eq:appendix-grover-zero-minus}
\end{equation}
Subtracting a multiple of this eigenket leaves $h=0$.

For either sign, all eigenkets with $h=0$ are specified by
\begin{equation}
 \psi_R(x+1,y)=z\psi_L(x,y),\qquad
 \psi_U(x,y+1)=z\psi_D(x,y),\qquad
 \psi_R+\psi_L-\psi_U-\psi_D=0,
 \label{eq:appendix-grover-flat-eigenvectors}
\end{equation}
where the first two relations apply to interior bonds and the last holds
at each site. The inward-pointing wall amplitudes
$\psi_R(0,y)$, $\psi_L(L_x-1,y)$, $\psi_U(x,0)$ and $\psi_D(x,L_y-1)$
are free for $z=+1$ and vanish for $z=-1$, by
\eqref{eq:appendix-grover-walls}. These relations construct the flat
eigenkets through \eqref{eq:appendix-grover-amplitudes}; at $-1$, add
$\ket{\Psi_0^-}$. A normalized uniform example at $+1$ is
$\ket{\Psi_0^+}=\ket{s}\otimes N^{-1/2}\sum_{x,y}\ket{x,y}$.

There is one free amplitude per interior bond, giving
$E=2N-L_x-L_y$ variables, and, at $+1$, one per inward-pointing wall
state, giving $B=2L_x+2L_y$ more. At $+1$, the $N$ conditions $h=0$
are independent: a vanishing linear combination of them must have zero
coefficients at boundary sites, because each wall variable occurs in only
one condition. Each bond then forces the coefficients at its two endpoints
to sum to zero, propagating the zero coefficients to every site.
At $-1$, each bond amplitude enters its endpoint conditions with opposite
signs. Their sum vanishes, and a spanning tree gives $N-1$ independent
conditions. Including the additional eigenket $\ket{\Psi_0^-}$ gives
\begin{equation}
 \operatorname{mult}(+1)=E+B-N=N+L_x+L_y,\qquad
 \operatorname{mult}(-1)=E-(N-1)+1=N-L_x-L_y+2.
 \label{eq:appendix-grover-multiplicities}
\end{equation}
Together with $z_{mn}^{\pm}$ for each $(m,n)\neq(0,0)$, these are the
complete spectrum. Coincident $a_{mn}$ values give dispersive degeneracies
without losing independent eigenkets: their scalar modes are independent,
and \eqref{eq:appendix-grover-components} reconstructs each uniquely.
The total dimension is
$(N+L_x+L_y)+(N-L_x-L_y+2)+2(N-1)=4N$, also for $L_x=1$ or $L_y=1$.
