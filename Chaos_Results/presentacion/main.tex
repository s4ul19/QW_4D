\section{Modelo y monedas}
\begin{frame}[t]{Caminata cuántica en billares}
  \captionline{\Large El modelo a trabajar: }
  \vspace{2mm}
  \[
    \mathcal H=\mathcal H_p\otimes\mathcal H_c,
    \qquad \dim\mathcal H_p=N,\quad \dim\mathcal H_c=4
  \]
  \[
    \ket{\psi_t}=\sum_{v\in\Omega}\sum_{c=1}^{4}
      \psi_c(v,t)\ket{v}\otimes\ket{c}
  \]
  \[
    \boxed{\ket{\psi_{t+1}}=U_{evol}\ket{\psi_t},\qquad
      U_{evol}=S\bigl(\mathbb I_p\otimes C_4\bigr)}
  \]
  \begin{itemize}
    \item $C_4$ mezcla las cuatro direcciones de movimiento.
    \item $S$ desplaza al caminante e incorpora la reflexión en la frontera.
  \end{itemize}
\end{frame}

\section{Estadística espectral}
\subsection{Reducción a caminatas unidimensionales}
\begin{frame}[t]{Espectro de cada segmento con reflexión}
  \noindent Para un segmento de $n$ sitios y un bloque unitario
  \[
    C_1=e^{i\alpha}\begin{pmatrix}a&b\\-b^*&a^*\end{pmatrix},
    \qquad |a|^2+|b|^2=1,
  \]
  \noindent los $2(n-1)$ autovalores de banda y los dos aislados son
  \[
    \lambda_{m,\pm}=e^{i\alpha}e^{\pm i\omega_m},\qquad
    \cos\omega_m=|a|\cos\frac{\pi m}{n},\quad m=1,\ldots,n-1,
  \]
  \[
    \lambda_{\mathrm{ais},\pm}=e^{i\alpha}
    \left(\pm\sqrt{1-(\operatorname{Im}b)^2}+i\operatorname{Im}b\right).
  \]
  \captionline{El artículo compara esta descomposición con ambos espectros numéricos y reporta una discrepancia máxima del orden de $10^{-15}$.}
\end{frame}

  \[

\section{Entrelazamiento}
\begin{frame}[t]{Entrelazamiento: estado inicial localizado}
  \[
    \rho_c(t)=\operatorname{Tr}_p\ket{\psi_t}\bra{\psi_t},
    \qquad S_c(t)=-\operatorname{Tr}[\rho_c(t)\ln\rho_c(t)].
  \]
  \ComparePlots{EntanglementEntropy_Rect.pdf}{EntanglementEntropy_Sinai.pdf}
    {$80\times50$, $1000$ pasos.}{$L=120$, $R=90$, $1000$ pasos.}
  \captionline{La entropía usa logaritmos naturales. El máximo para la moneda de dimensión 4 es $\ln4\simeq1.386$.}
\end{frame}

\section{Conclusiones}
\begin{frame}[t]{Conclusiones}
  \begin{itemize}
    \setlength\itemsep{3.5mm}
    \item Las condiciones de frontera impuestas por la geometría no son necesarias para la emergencia de caos cuántico.
    \item En Sinai, \coin{g}, \coin{gpg} y \coin{oo} son compatibles con COE; \coin{b}, \coin{uoou}, \coin{o}, \coin{ubu}, \coin{u} y \coin{upu}, con CUE.
    \item En el rectángulo, los casos con caos cuántico parecen seguir una estadística COE.
    \item Las monedas que producen integrabilidad muestran eigenestados más localizados que las monedas caóticas.
    \item La mayoría de las monedas alcanza entropías cercanas a $\ln 4$; \coin{p} muestra caídas recurrentes.
    \item Mediante el IPR espacial no es concluyente si sea posible identificar dinámicas caóticas de no caóticas.
  \end{itemize}
\end{frame}

  \[

\section{Apéndice temporal}
\begin{frame}[t,label=ap-temporal]{Expresión del límite del promedio temporal}
  \noindent{\small Sea $\Pi_\lambda$ el proyector sobre el subespacio de eigenvalor $\lambda$ de $U$.}\par
  \[
    \lim_{T\to\infty}\frac1T\sum_{t=0}^{T-1}
      U^t\rho_0 U^{-t}
    =\sum_{\lambda}\Pi_\lambda\rho_0\Pi_\lambda,
    \qquad\rho_0=\ket{\psi_0}\bra{\psi_0}.
  \]
  \[
    \pi(v)=\sum_{c,\lambda}
      \left|\bra{v,c}\Pi_\lambda\ket{\psi_0}\right|^2.
  \]
  \captionline{El promedio elimina interferencias entre eigenvalores distintos. Conserva las coherencias dentro de los subespacios degenerados.}
  \captionline{Referencia de las notas originales: Aharonov et al., \emph{Quantum Walks on Graphs} (2001), arXiv:quant-ph/0012090.}
\end{frame}
